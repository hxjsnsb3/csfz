-- 状态标签
local statusLabel = MusicTab:AddLabel("当前状态: 未播放")

-- 修复的音乐ID验证函数
local function isValidMusicID(id)
    -- 检查是否为纯数字且长度在6-10位之间
    return id:match("^%d+$") and #id >= 6 and #id <= 10
end

-- 播放按钮
MusicTab:AddButton({
    Name = "播放音乐",
    Callback = function()
        -- 获取输入框的值
        local musicId = musicIdInput.Value
        
        -- 清理非数字字符
        musicId = musicId:gsub("%D+", "")
        
        -- 验证音乐ID
        if not isValidMusicID(musicId) then
            statusLabel:Set("状态: 请输入有效的音乐ID (6-10位数字)")
            return
        end
        
        -- 停止当前音乐
        if currentSound then
            currentSound:Stop()
            currentSound:Destroy()
            currentSound = nil
        end
        
        -- 创建新音乐
        currentSound = Instance.new("Sound")
        currentSound.SoundId = "rbxassetid://" .. musicId
        currentSound.Volume = 5
        currentSound.Parent = game:GetService("SoundService")
        
        statusLabel:Set("状态: 加载中...")
        
        -- 处理加载完成
        local connection
        connection = currentSound.Loaded:Connect(function()
            if currentSound and currentSound.IsLoaded then
                connection:Disconnect()
                currentSound:Play()
                statusLabel:Set("状态: 正在播放 - " .. musicId)
                
                -- 播放结束时更新状态
                currentSound.Ended:Connect(function()
                    statusLabel:Set("状态: 播放完成 - " .. musicId)
                    wait(2)
                    statusLabel:Set("状态: 未播放")
                end)
            else
                connection:Disconnect()
                statusLabel:Set("状态: 加载失败! 请检查音乐ID")
                if currentSound then
                    currentSound:Destroy()
                    currentSound = nil
                end
            end
        end)
        
        -- 添加超时检测 (5秒)
        spawn(function()
            wait(5)
            if currentSound and not currentSound.IsLoaded then
                connection:Disconnect()
                statusLabel:Set("状态: 加载超时! 请重试")
                currentSound:Destroy()
                currentSound = nil
            end
        end)
    end
})

-- 停止按钮
MusicTab:AddButton({

        if currentSound then
            currentSound:Stop()
            currentSound:Destroy()

            statusLabel:Set("状态: 已停止")
        else
            statusLabel:Set("状态: 没有正在播放的音乐")
        end
    end
})
